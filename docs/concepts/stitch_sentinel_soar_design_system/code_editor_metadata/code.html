import requests import json from sentinel_sdk import Context, Alert def enrich_ip(ip_address: str) -> dict: """Queries VirusTotal for IP reputation.""" api_key = Context.get_secret('VT_API_KEY') url = f"https://www.virustotal.com/api/v3/ip_addresses/{ip_address}" headers = { "accept": "application/json", "x-apikey": api_key } response = requests.get(url, headers=headers) if response.status_code == 200: data = response.json() stats = data.get('data', {}).get('attributes', {}).get('last_analysis_stats', {}) return stats else: return {"error": f"VT API Error: {response.status_code}"} def main(alert: Alert): # Extract IPs from the incoming alert target_ips = alert.get_observables(type='ip') for ip in target_ips: results = enrich_ip(ip) # Evaluate malicious threshold malicious_count = results.get('malicious', 0) if malicious_count > 5: alert.add_tag("High_Severity_IP") alert.escalate(reason=f"IP {ip} has {malicious_count} malicious hits.") # Attach raw data to alert context alert.add_context(f"VT_Report_{ip}", results) return True
